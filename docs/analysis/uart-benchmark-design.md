**新增文件**: `kernel/src/drivers/benchmark.rs`

```rust
//! UART 性能测试模块
//!
//! 提供内核态性能统计和测试接口

use core::sync::atomic::{AtomicU64, AtomicBool, Ordering};
use axhal::time::monotonic_time_nanos;

// 统计计数器
static TX_BYTES: AtomicU64 = AtomicU64::new(0);
static RX_BYTES: AtomicU64 = AtomicU64::new(0);
static IRQ_COUNT: AtomicU64 = AtomicU64::new(0);
static START_TIME: AtomicU64 = AtomicU64::new(0);
static BENCHMARK_ACTIVE: AtomicBool = AtomicBool::new(false);

/// 开始基准测试
pub fn start() {
    START_TIME.store(monotonic_time_nanos(), Ordering::Relaxed);
    TX_BYTES.store(0, Ordering::Relaxed);
    RX_BYTES.store(0, Ordering::Relaxed);
    IRQ_COUNT.store(0, Ordering::Relaxed);
    BENCHMARK_ACTIVE.store(true, Ordering::Relaxed);
    ax_println!("[BENCH] Started");
}

/// 停止基准测试并报告结果
pub fn stop() {
    BENCHMARK_ACTIVE.store(false, Ordering::Relaxed);
    report();
}

/// 记录 TX 字节
pub fn record_tx(bytes: u64) {
    if BENCHMARK_ACTIVE.load(Ordering::Relaxed) {
        TX_BYTES.fetch_add(bytes, Ordering::Relaxed);
    }
}

/// 记录 RX 字节
pub fn record_rx(bytes: u64) {
    if BENCHMARK_ACTIVE.load(Ordering::Relaxed) {
        RX_BYTES.fetch_add(bytes, Ordering::Relaxed);
    }
}

/// 记录 IRQ
pub fn record_irq() {
    if BENCHMARK_ACTIVE.load(Ordering::Relaxed) {
        IRQ_COUNT.fetch_add(1, Ordering::Relaxed);
    }
}

/// 报告测试结果
pub fn report() {
    let elapsed = monotonic_time_nanos() - START_TIME.load(Ordering::Relaxed);
    let tx = TX_BYTES.load(Ordering::Relaxed);
    let rx = RX_BYTES.load(Ordering::Relaxed);
    let irq = IRQ_COUNT.load(Ordering::Relaxed);

    ax_println!("=== UART Benchmark Report ===");
    ax_println!("Elapsed: {:.2} ms", elapsed as f64 / 1_000_000.0);
    ax_println!("TX: {} bytes ({:.2} KB/s)", tx, tx as f64 * 1_000_000_000.0 / elapsed as f64 / 1024.0);
    ax_println!("RX: {} bytes ({:.2} KB/s)", rx, rx as f64 * 1_000_000_000.0 / elapsed as f64 / 1024.0);
    ax_println!("IRQs: {} ({:.2}/s)", irq, irq as f64 * 1_000_000_000.0 / elapsed as f64);
    ax_println!("=============================");
}

/// 内存使用统计
pub fn memory_usage() {
    let rx_buf = 64 * 1024; // BUF_SIZE
    let tx_buf = 64 * 1024; // BUF_SIZE
    let driver = core::mem::size_of::<crate::drivers::async_driver::AsyncUartDriver>();

    ax_println!("=== Memory Usage ===");
    ax_println!("RX Buffer: {} KB", rx_buf / 1024);
    ax_println!("TX Buffer: {} KB", tx_buf / 1024);
    ax_println!("Driver Struct: {} bytes", driver);
    ax_println!("Total: {} KB", (rx_buf + tx_buf + driver) / 1024);
    ax_println!("====================");
}
```

### 4.3 用户态测试程序

**新增文件**: `tests/benchmark.c`

```c
/**
 * UART 异步串口性能测试程序
 *
 * 测试项目:
 * 1. TX 吞吐量
 * 2. RX 吞吐量
 * 3. 延迟 (单字节 echo)
 * 4. 数据完整性
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <time.h>
#include <sys/stat.h>

#define DEVICE_PATH "/dev/console"
#define BUF_SIZE 1024
#define THROUGHPUT_TEST_SIZE (1024 * 1024)  // 1MB
#define LATENCY_ITERATIONS 100

// 吞吐量测试
void test_throughput() {
    printf("=== Throughput Test ===\n");

    int fd = open(DEVICE_PATH, O_WRONLY);
    if (fd < 0) {
        perror("open");
        return;
    }

    char *buf = malloc(BUF_SIZE);
    memset(buf, 'A', BUF_SIZE);

    struct timespec start, end;
    clock_gettime(CLOCK_MONOTONIC, &start);

    size_t total = 0;
    while (total < THROUGHPUT_TEST_SIZE) {
        ssize_t n = write(fd, buf, BUF_SIZE);
        if (n > 0) total += n;
    }

    clock_gettime(CLOCK_MONOTONIC, &end);

    double elapsed = (end.tv_sec - start.tv_sec) +
                     (end.tv_nsec - start.tv_nsec) / 1e9;
    double throughput = total / elapsed / 1024.0;

    printf("Sent: %zu bytes\n", total);
    printf("Time: %.3f s\n", elapsed);
    printf("Throughput: %.2f KB/s\n", throughput);
    printf("Line rate: %.1f%%\n", throughput / 11.52 * 100);

    free(buf);
    close(fd);
}

// 延迟测试
void test_latency() {
    printf("=== Latency Test ===\n");

    int fd = open(DEVICE_PATH, O_RDWR);
    if (fd < 0) {
        perror("open");
        return;
    }

    long latencies[LATENCY_ITERATIONS];

    for (int i = 0; i < LATENCY_ITERATIONS; i++) {
        struct timespec start, end;
        clock_gettime(CLOCK_MONOTONIC, &start);

        char tx = 'A' + (i % 26);
        char rx;
        write(fd, &tx, 1);
        read(fd, &rx, 1);

        clock_gettime(CLOCK_MONOTONIC, &end);
        latencies[i] = (end.tv_sec - start.tv_sec) * 1000000000 +
                       (end.tv_nsec - start.tv_nsec);
    }

    // 计算统计值
    long sum = 0, min = latencies[0], max = latencies[0];
    for (int i = 0; i < LATENCY_ITERATIONS; i++) {
        sum += latencies[i];
        if (latencies[i] < min) min = latencies[i];
        if (latencies[i] > max) max = latencies[i];
    }

    // 排序计算百分位
    // ... (简化实现)

    printf("Iterations: %d\n", LATENCY_ITERATIONS);
    printf("Min: %ld ns (%.3f ms)\n", min, min / 1e6);
    printf("Max: %ld ns (%.3f ms)\n", max, max / 1e6);
    printf("Avg: %ld ns (%.3f ms)\n", sum / LATENCY_ITERATIONS,
           (sum / LATENCY_ITERATIONS) / 1e6);

    close(fd);
}

int main() {
    printf("UART Async Benchmark\n");
    printf("====================\n\n");

    test_throughput();
    printf("\n");
    test_latency();

    return 0;
}
```

### 4.4 自动化测试脚本

**新增文件**: `scripts/benchmark.sh`

```bash
#!/bin/bash
# UART 异步串口性能测试脚本

set -e

QEMU_PORT=4444
QEMU_PID=""
RESULTS_DIR="benchmark_results"

# 清理函数
cleanup() {
    if [ -n "$QEMU_PID" ]; then
        kill $QEMU_PID 2>/dev/null || true
    fi
}

trap cleanup EXIT

# 构建内核
build_kernel() {
    echo "Building kernel..."
    make build
}

# 启动 QEMU
start_qemu() {
    echo "Starting QEMU..."
    make run QEMU_ARGS="-monitor none -serial tcp::${QEMU_PORT},server=on" &
    QEMU_PID=$!

    # 等待 QEMU 启动
    sleep 2

    # 等待 Shell 就绪
    echo "Waiting for shell..."
    timeout 10 bash -c "until echo '' | nc -q 1 localhost ${QEMU_PORT} 2>/dev/null | grep -q 'starry:~#'; do sleep 0.5; done"
}

# 执行测试命令
run_test() {
    local cmd="$1"
    echo "$cmd" | nc -q 1 localhost ${QEMU_PORT}
}

# 运行基准测试
run_benchmark() {
    echo "Running benchmark..."

    # 启动内核态基准测试
    run_test "bench start"

    # 运行用户态测试
    run_test "/tmp/benchmark"

    # 停止并获取报告
    run_test "bench stop"
}

# 主函数
main() {
    mkdir -p ${RESULTS_DIR}

    build_kernel
    start_qemu
    run_benchmark

    echo "Results saved to ${RESULTS_DIR}/"
}

main "$@"
```

## 7. 注意事项

### 7.1 QEMU 限制

- QEMU 的串口模拟不完全等同于真实硬件
- 时钟精度有限，延迟测量可能有误差
- 建议在真板 (VisionFive2) 上验证关键指标

### 7.2 测试环境

- 确保 QEMU 使用 `-nographic` 模式
- 使用 TCP 串口连接便于自动化
- 测试期间避免其他进程干扰

### 7.3 结果解读

- 吞吐量目标 90% 线速是合理的（考虑协议开销）
- 延迟 P99 < 2ms 是可接受的（考虑 QEMU 调度）
- 内存占用 128KB 是固定的（Ring Buffer 配置）

## 8. 后续优化方向

根据测试结果，可能的优化方向：

1. **如果吞吐量不达标**:
   - 调整 NAPI_THRESHOLD 和 NAPI_BATCH_SIZE
   - 优化 Ring Buffer 大小
   - 减少锁竞争

2. **如果延迟过高**:
   - 优化 ISR 处理时间
   - 减少 copier 任务调度延迟
   - 使用更高效的唤醒机制

3. **如果内存占用过高**:
   - 减小 Ring Buffer 大小
   - 使用动态分配策略
   - 优化数据结构布局
